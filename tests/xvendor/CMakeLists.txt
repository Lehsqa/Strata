# These hardware feasibility probes are invoked manually, never registered with ctest.
if(CMAKE_SOURCE_DIR STREQUAL CMAKE_CURRENT_SOURCE_DIR)
    cmake_minimum_required(VERSION 3.24)
    project(strata_xvendor LANGUAGES CXX)
endif()

if(NOT CMAKE_SYSTEM_NAME STREQUAL "Linux")
    message(FATAL_ERROR "Cross-vendor feasibility probes require Linux memfd/futex")
endif()

find_package(Threads REQUIRED)
add_executable(strata-xvendor-shm-test shm_region_test.cpp)
target_compile_features(strata-xvendor-shm-test PRIVATE cxx_std_17)
target_link_libraries(strata-xvendor-shm-test PRIVATE Threads::Threads)

add_executable(strata-xvendor-inproc xvendor_inproc.cpp)
target_compile_features(strata-xvendor-inproc PRIVATE cxx_std_17)
target_link_libraries(strata-xvendor-inproc PRIVATE ${CMAKE_DL_LIBS})
add_executable(strata-xvendor-cli-test xvendor_cli_test.cpp)
target_compile_features(strata-xvendor-cli-test PRIVATE cxx_std_17)
foreach(_target strata-xvendor-shm-test strata-xvendor-inproc strata-xvendor-cli-test)
    if(TARGET strata_warnings)
        target_link_libraries(${_target} PRIVATE strata_warnings)
    else()
        target_compile_options(${_target} PRIVATE -Wall -Wextra -Wpedantic)
        if(STRATA_WERROR)
            target_compile_options(${_target} PRIVATE -Werror)
        endif()
    endif()
endforeach()

if(TARGET strata_kernels AND (STRATA_ENABLE_CUDA OR STRATA_ENABLE_HIP))
    add_executable(strata-xvendor-handoff xvendor_handoff.cu)
    target_include_directories(strata-xvendor-handoff PRIVATE ${PROJECT_SOURCE_DIR}/include ${_strata_gpu_include_directories})
    target_link_libraries(strata-xvendor-handoff PRIVATE strata_kernels ${_strata_gpu_runtime_target} Threads::Threads)
    target_compile_features(strata-xvendor-handoff PRIVATE cxx_std_17)
    set_target_properties(strata-xvendor-handoff PROPERTIES CUDA_STANDARD 17 CUDA_STANDARD_REQUIRED ON HIP_STANDARD 17 HIP_STANDARD_REQUIRED ON)
    if(STRATA_ENABLE_HIP OR STRATA_HIP_GFX906)
        set_source_files_properties(xvendor_handoff.cu PROPERTIES LANGUAGE HIP)
        set(_probe_vendor hip)
    else()
        set(_probe_vendor cuda)
        set_target_properties(strata-xvendor-handoff PROPERTIES CUDA_RUNTIME_LIBRARY Shared)
    endif()
    add_library(xprobe_${_probe_vendor} SHARED xprobe.cpp)
    target_compile_features(xprobe_${_probe_vendor} PRIVATE cxx_std_17)
    target_include_directories(xprobe_${_probe_vendor} PRIVATE ${_strata_gpu_include_directories})
    target_link_libraries(xprobe_${_probe_vendor} PRIVATE ${_strata_gpu_runtime_target})
    target_compile_options(xprobe_${_probe_vendor} PRIVATE -fvisibility=hidden)
    target_link_options(xprobe_${_probe_vendor} PRIVATE -Wl,--exclude-libs,ALL -Wl,-Bsymbolic
        "-Wl,--version-script=${CMAKE_CURRENT_SOURCE_DIR}/xprobe.exports")
    set_target_properties(xprobe_${_probe_vendor} PROPERTIES LINK_DEPENDS "${CMAKE_CURRENT_SOURCE_DIR}/xprobe.exports")
    if(TARGET strata_warnings)
        target_link_libraries(strata-xvendor-handoff PRIVATE strata_warnings)
        target_link_libraries(xprobe_${_probe_vendor} PRIVATE strata_warnings)
    endif()
endif()
